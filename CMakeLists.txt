cmake_minimum_required(VERSION 3.20)
project(Pi4AoIP VERSION 2.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Pi4-AoIP requires Linux; the runtime additionally checks Pi 4 and PREEMPT_RT")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(cmake/AoIPDependency.cmake)
add_library(aoip_rpi4 STATIC src/rpi4.cpp)
add_library(Pi4AoIP::platform ALIAS aoip_rpi4)
add_library(PiAoIP::rpi4 ALIAS aoip_rpi4)
set_target_properties(aoip_rpi4 PROPERTIES EXPORT_NAME platform POSITION_INDEPENDENT_CODE ON)
target_include_directories(aoip_rpi4 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(aoip_rpi4 PUBLIC AoIP::peer)
add_executable(aoip_peer_rpi4 apps/main.cpp)
target_link_libraries(aoip_peer_rpi4 PRIVATE Pi4AoIP::platform)
set_target_properties(aoip_peer_rpi4 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(rpi4_host_info examples/host_info.cpp)
target_link_libraries(rpi4_host_info PRIVATE Pi4AoIP::platform)
enable_testing()
add_test(NAME host_info COMMAND rpi4_host_info)
install(TARGETS aoip_rpi4 EXPORT Pi4AoIPTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT Pi4AoIPTargets NAMESPACE Pi4AoIP:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Pi4AoIP)
configure_package_config_file(cmake/Pi4AoIPConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/Pi4AoIPConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Pi4AoIP)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/Pi4AoIPConfigVersion.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/Pi4AoIPConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/Pi4AoIPConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Pi4AoIP)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/Pi4-AoIP)
